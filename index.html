// E-Mail Login (Magic Link) mit erweiterter Fehlerprüfung async function loginWithEmail() { const emailInput = document.getElementById('email'); if (!emailInput) { alert("Eingabefeld nicht gefunden."); return; } const email = emailInput.value.trim(); if (!email || !email.includes('@')) { alert("Bitte gib eine gültige E-Mail-Adresse ein (z. B. name@firma.de)."); return; } try { const { error } = await db.auth.signInWithOtp({ email: email, options: { emailRedirectTo: window.location.href } }); if (error) { alert("Fehler bei der Anmeldung: " + error.message); } else { alert("Anmeldelink wurde erfolgreich an " + email + " gesendet! Bitte prüfe dein E-Mail-Postfach."); } } catch (err) { alert("Ein unerwarteter Fehler ist aufgetreten: " + err.message); } }
